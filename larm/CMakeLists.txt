cmake_minimum_required(VERSION 3.25...3.31)
project(larm VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(LARM_WARNINGS_AS_ERRORS "Treat compiler warnings in larm targets as errors" OFF)
option(LARM_BUILD_TESTS "Build the larm test suite" ${PROJECT_IS_TOP_LEVEL})
option(LARM_WITH_MUJOCO "Build the MuJoCo simulation backend and its tools" ON)
option(LARM_WITH_ROS "Build the ROS 2 node; needs a sourced ROS 2 environment with larm_msgs" ON)
option(LARM_WITH_STUDIO "Build larm studio, with ROS and MuJoCo, when Qt 6 is found" ON)
option(LARM_WITH_ROBSTRIDE "Build the SocketCAN RobStride driver (Linux)" ON)
set(LARM_MUJOCO_ROOT "" CACHE PATH "Official MuJoCo release directory containing include/ and lib/")
set(LARM_ROBOT_PROFILE
    "${CMAKE_CURRENT_SOURCE_DIR}/../robots/rebot_b601/config/rebot_b601_rs.yaml"
    CACHE FILEPATH "Robot profile used by the integration tests")

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(LarmCompilerWarnings)

if(LARM_WITH_ROS)
    find_package(ament_cmake REQUIRED)
endif()
if(LARM_BUILD_TESTS)
    enable_testing()
    find_package(GTest REQUIRED)
    include(GoogleTest)
    include(LarmTesting)
endif()
include(LarmExecDeps)
if(LARM_WITH_MUJOCO)
    include(LarmMujoco)
endif()

add_subdirectory(core)
add_subdirectory(model)
add_subdirectory(motion)
add_subdirectory(hal)
add_subdirectory(control)
if(LARM_WITH_ROBSTRIDE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(drivers/can)
endif()
if(LARM_WITH_MUJOCO)
    add_subdirectory(sim)
    add_subdirectory(apps/sim_cli)
endif()
add_subdirectory(runtime)
if(TARGET larm::can)
    add_subdirectory(drivers/robstride)
    add_subdirectory(apps/driver_probe)
endif()
if(LARM_WITH_ROS)
    add_subdirectory(ros)
    if(LARM_WITH_MUJOCO)
        add_subdirectory(apps/runtime_node)
    endif()
endif()
if(LARM_WITH_STUDIO AND LARM_WITH_ROS AND LARM_WITH_MUJOCO)
    find_package(Qt6 6.2 QUIET COMPONENTS Widgets OpenGL)
    if(Qt6_FOUND)
        add_subdirectory(studio)
        add_subdirectory(apps/studio)
    else()
        message(STATUS "larm studio skipped: Qt 6 Widgets and OpenGL not found (add Qt to CMAKE_PREFIX_PATH)")
    endif()
endif()

if(LARM_BUILD_TESTS AND LARM_WITH_MUJOCO)
    add_subdirectory(tests)
endif()

if(LARM_WITH_ROS)
    ament_package()
endif()
